static void Filter_Init(void)
{
    for (int ch = 0; ch < N_CH; ch++)
    {
    	/* 50 Hz notch, fs=2000, Q=5 */
    	notch50[ch].b[0] = 0.9845975f;
    	notch50[ch].b[1] = -1.9449509f;
    	notch50[ch].b[2] = 0.9845975f;

    	notch50[ch].a[0] = 1.0f;
    	notch50[ch].a[1] = -1.9449509f;
    	notch50[ch].a[2] = 0.9691950f;

    	/* 20 Hz highpass, fs=2000, Butterworth Q=0.70710678 */
    	highpass20[ch].b[0] = 0.9565432f;
    	highpass20[ch].b[1] = -1.9130865f;
    	highpass20[ch].b[2] = 0.9565432f;

    	highpass20[ch].a[0] = 1.0f;
    	highpass20[ch].a[1] = -1.9111971f;
    	highpass20[ch].a[2] = 0.9149758f;

    	// 在 Filter_Init() 中替换原有的 lowpass400 初始化代码：

    	/* 400 Hz Lowpass, 4th-Order Butterworth, Stage 1 (Q=0.5412) */
    	lowpass400_s1[ch].b[0] = 0.183921f;
    	lowpass400_s1[ch].b[1] = 0.367842f;
    	lowpass400_s1[ch].b[2] = 0.183921f;
    	lowpass400_s1[ch].a[0] = 1.0f;
    	lowpass400_s1[ch].a[1] = -0.329007f;
    	lowpass400_s1[ch].a[2] = 0.064691f;

    	/* 400 Hz Lowpass, 4th-Order Butterworth, Stage 2 (Q=1.3065) */
    	lowpass400_s2[ch].b[0] = 0.253331f;
    	lowpass400_s2[ch].b[1] = 0.506662f;
    	lowpass400_s2[ch].b[2] = 0.253331f;
    	lowpass400_s2[ch].a[0] = 1.0f;
    	lowpass400_s2[ch].a[1] = -0.453228f;
    	lowpass400_s2[ch].a[2] = 0.466551f;

    	/* 别忘了把它们的状态变量 x[i] 和 y[i] 都在下方循环中初始化为 0.0f */

        /* 初始化所有状态变量为 0 */
        for (int i = 0; i < 3; i++)
        {
            notch50[ch].x[i] = 0.0f;
            notch50[ch].y[i] = 0.0f;

            highpass20[ch].x[i] = 0.0f;
            highpass20[ch].y[i] = 0.0f;

            // 新增：把第一级 (s1) 的输入输出历史清零
            lowpass400_s1[ch].x[i] = 0.0f;
            lowpass400_s1[ch].y[i] = 0.0f;

                        // 新增：把第二级 (s2) 的输入输出历史清零
            lowpass400_s2[ch].x[i] = 0.0f;
            lowpass400_s2[ch].y[i] = 0.0f;
        }
    }
}

static inline int u16_to_ascii(uint16_t v, char *dst)
{
    char tmp[5];
    int n = 0;
    int k = 0;

    if (v == 0)
    {
        dst[0] = '0';
        return 1;
    }

    while (v > 0)
    {
        tmp[n++] = (char)('0' + (v % 10));
        v /= 10;
    }

    while (n > 0)
    {
        dst[k++] = tmp[--n];
    }

    return k;
}

static inline int append_u16_csv4_line(char *dst, int remaining,
                                       uint16_t a, uint16_t b, uint16_t c, uint16_t d)
{
    int pos = 0;
    int n;

    n = u16_to_ascii(a, &dst[pos]);
    pos += n;
    if (pos >= remaining) return -1;
    dst[pos++] = ',';

    n = u16_to_ascii(b, &dst[pos]);
    pos += n;
    if (pos >= remaining) return -1;
    dst[pos++] = ',';

    n = u16_to_ascii(c, &dst[pos]);
    pos += n;
    if (pos >= remaining) return -1;
    dst[pos++] = ',';

    n = u16_to_ascii(d, &dst[pos]);
    pos += n;
    if (pos >= remaining) return -1;
    dst[pos++] = '\n';

    if (pos > remaining) return -1;
    return pos;
}

static inline float Biquad_Process(BiquadFilter *f, float input)
{
    f->x[2] = f->x[1];
    f->x[1] = f->x[0];
    f->x[0] = input;

    f->y[2] = f->y[1];
    f->y[1] = f->y[0];

    f->y[0] = (f->b[0] * f->x[0]) + (f->b[1] * f->x[1]) + (f->b[2] * f->x[2])
            - (f->a[1] * f->y[1]) - (f->a[2] * f->y[2]);

    return f->y[0];
}
#define ADC_MID 2048.0f

static void ProcessBlock(const uint16_t* in, uint16_t* out, int frames)
{
    for (int i = 0; i < frames; i++)
    {
        for (int ch = 0; ch < N_CH; ch++)
        {
            float raw = (float)in[i * N_CH + ch];

            /* 先减去直流中点，变成零均值信号 */
            float x0 = raw - ADC_MID;

            /* 1. 20Hz 高通滤波：滤除电极片晃动和极低频漂移 */
            float hp_out = Biquad_Process(&highpass20[ch], x0);

            /* 2. 50Hz 陷波滤波：滤除电网工频干扰 */
            float notch_out = Biquad_Process(&notch50[ch], hp_out);

            /* 3. 400Hz 四阶低通滤波 (级联两级) */
            float lp_out_s1 = Biquad_Process(&lowpass400_s1[ch], notch_out);
            float lp_out_s2 = Biquad_Process(&lowpass400_s2[ch], lp_out_s1); // 把 s1 的结果喂给 s2

            /* 再加回中值用于图表显示或上位机传输 */
            float final_f = lp_out_s2 + ADC_MID;

            /* 防止溢出截断 */
            if (final_f > 4095.0f) final_f = 4095.0f;
            if (final_f < 0.0f)    final_f = 0.0f;

            out[i * N_CH + ch] = (uint16_t)final_f;
        }
    }
}
static void TryUartSendCsvBlock(const uint16_t* out, int frames)
{
    if (uart_tx_busy) return;

    uint8_t sel = (uint8_t)(tx_sel ^ 1U);
    tx_sel = sel;

    int pos = 0;

    for (int i = 0; i < frames; i++)
    {
        int remaining = (int)TX_BUF_SIZE - pos;
        if (remaining < 32) break;

        int n = append_u16_csv4_line((char*)&tx_buf[sel][pos], remaining,
                                     out[i*N_CH + 0],
                                     out[i*N_CH + 1],
                                     out[i*N_CH + 2],
                                     out[i*N_CH + 3]);
        if (n < 0) break;

        pos += n;
    }

    if (pos <= 0) return;

    uart_tx_busy = 1;

    if (HAL_UART_Transmit_DMA(&huart2, tx_buf[sel], (uint16_t)pos) != HAL_OK)
    {
        uart_tx_busy = 0;
    }
}

    while (1)

    /* USER CODE BEGIN 3 */

      if (adc_half_ready)
      {
          adc_half_ready = 0;

          /* 前半段 block 的起始地址 */
          const uint16_t* p = &adc_dma_buf[0];

          ProcessBlock(p, filt_block, FRAMES_PER_BLOCK);

          if (warmup_left > 0U)
          {
              warmup_left--;
          }
          else
          {
              TryUartSendCsvBlock(filt_block, FRAMES_PER_BLOCK);
          }

      }

      if (adc_full_ready)
      {
          adc_full_ready = 0;

          /* 后半段 block 的起始地址 */
          const uint16_t* p = &adc_dma_buf[ADC_HALF_LEN];

          ProcessBlock(p, filt_block, FRAMES_PER_BLOCK);

          {
              warmup_left--;
          }
          else
          {
              TryUartSendCsvBlock(filt_block, FRAMES_PER_BLOCK);
          }

      }

      /* 空闲就睡，等待 DMA 中断唤醒 */
#ifndef NDEBUG
  /* Debug build，先不睡，方便调试 */
#else
  if (!adc_half_ready && !adc_full_ready)
  {
    __WFI();
  }
#endif

    }
  /* USER CODE END 3 */
}

/* ADC DMA 半传输完成回调：表示前半段缓冲区填满 */
void HAL_ADC_ConvHalfCpltCallback(ADC_HandleTypeDef* hadc)
{
    if (hadc->Instance == ADC1)
    {
        if (adc_half_ready)
        {
            adc_half_overrun++;
        }
        adc_half_ready = 1;
    }
}


/* ADC DMA 全传输完成回调：表示后半段缓冲区填满 */
void HAL_ADC_ConvCpltCallback(ADC_HandleTypeDef* hadc)
{
    if (hadc->Instance == ADC1)
    {
        if (adc_full_ready)
        {
            adc_full_overrun++;
        }
        adc_full_ready = 1;
    }
}
